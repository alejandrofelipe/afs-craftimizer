# UI Studio — Plano de Implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Criar `Craftimizer.UI` (shared library sem Dalamud) e `Craftimizer.UIStudio` (app Storybook standalone) para visualizar componentes de UI visualmente sem abrir o FFXIV.

**Architecture:** `Craftimizer/Utils/UI/*.cs` (exceto `SqText.cs` e `DynamicBars.cs`) migram para `Craftimizer.UI.csproj` que depende apenas de `ImGui.NET`. Dependências Dalamud-runtime (`GlobalScale`, `IconFont`) são isoladas via `IUiServices` + singleton `UiServices.Current`. O Studio usa Silk.NET + ImGui.NET com FontAwesome carregado manualmente e implementa Storybook com discovery automático de stories por reflection.

**Tech Stack:** .NET 10, ImGui.NET 1.90.9.1, Silk.NET.OpenGL + Silk.NET.Windowing + Silk.NET.Input, FontAwesome Free 6 TTF

**Spec:** `docs/superpowers/specs/2026-06-09-ui-studio-design.md`

> **Nota de escopo:** Os seguintes arquivos ficam no plugin (dependências de game data ou ImPlot):
> - `SqText.cs` — usa `Dalamud.Game.Text.SeIconChar`
> - `DynamicBars.cs` — usa `SimulatedMacro` do Simulator
> - `ImGuiUtils.Charts.cs` é **dividido**: só `DrawStatArc` (puro ImGui) move para `Craftimizer.UI`; `DrawMacroStatArcs`, `ViolinPlot` e `ViolinData` ficam no plugin (usam `SimulationState` + `Dalamud.Bindings.ImPlot` + `MathNet`).

```csharp
// Craftimizer.UI/ImGuiUtils.Charts.cs
using System.Numerics;

namespace Craftimizer.Utils;

internal static partial class ImGuiUtils
{
    private static float Lerp(float a, float b, float t) =>
        MathF.FusedMultiplyAdd(b - a, t, a);

    // Draws a 280° arc (gap at bottom) into drawList.
    // screenPos = top-left of bounding square; size = side length; frac = 0..1 fill.
    public static void DrawStatArc(ImDrawListPtr drawList, Vector2 screenPos, float size, float frac, Vector4 color)
    {
        const float StartAngle = 2.269f;
        const float SweepAngle = 4.887f;

        var center  = screenPos + new Vector2(size * 0.5f, size * 0.5f);
        var strokeW = MathF.Max(2f, size * 0.16f);
        var radius  = size * 0.5f - strokeW * 0.5f - 1f;
        var capR    = strokeW * 0.5f;

        var trackColor = ImGui.GetColorU32(color with { W = 0.20f });
        var fillColor  = ImGui.GetColorU32(color);

        drawList.PathArcTo(center, radius, StartAngle, StartAngle + SweepAngle, 32);
        drawList.PathStroke(trackColor, ImDrawFlags.None, strokeW);
        drawList.AddCircleFilled(center + new Vector2(MathF.Cos(StartAngle)              * radius, MathF.Sin(StartAngle)              * radius), capR, trackColor);
        drawList.AddCircleFilled(center + new Vector2(MathF.Cos(StartAngle + SweepAngle) * radius, MathF.Sin(StartAngle + SweepAngle) * radius), capR, trackColor);

        if (frac > 0.005f)
        {
            var fillEnd = StartAngle + SweepAngle * MathF.Min(frac, 1f);
            drawList.PathArcTo(center, radius, StartAngle, fillEnd, 32);
            drawList.PathStroke(fillColor, ImDrawFlags.None, strokeW);
            drawList.AddCircleFilled(center + new Vector2(MathF.Cos(StartAngle) * radius, MathF.Sin(StartAngle) * radius), capR, fillColor);
            drawList.AddCircleFilled(center + new Vector2(MathF.Cos(fillEnd)    * radius, MathF.Sin(fillEnd)    * radius), capR, fillColor);
        }
    }
}
```

```csharp
// UIStudio/ImGuiController.cs
using ImGuiNET;
using Silk.NET.Input;
using Silk.NET.OpenGL;
using Silk.NET.Windowing;
using System.Numerics;
using System.Runtime.InteropServices;

namespace Craftimizer.UIStudio;

internal sealed unsafe class ImGuiController : IDisposable
{
    private readonly GL _gl;
    private readonly IWindow _window;

    private uint _vao, _vbo, _ebo;
    private uint _shader;
    private uint _fontTexture;

    private int _uniformTex;
    private int _uniformProjMtx;
    private uint _attribVtxPos;
    private uint _attribVtxUV;
    private uint _attribVtxColor;

    private const string VertSrc = @"
        #version 330 core
        layout(location=0) in vec2 Position;
        layout(location=1) in vec2 UV;
        layout(location=2) in vec4 Color;
        uniform mat4 ProjMtx;
        out vec2 Frag_UV; out vec4 Frag_Color;
        void main(){Frag_UV=UV;Frag_Color=Color;gl_Position=ProjMtx*vec4(Position,0,1);}";

    private const string FragSrc = @"
        #version 330 core
        in vec2 Frag_UV; in vec4 Frag_Color;
        uniform sampler2D Texture;
        layout(location=0) out vec4 Out_Color;
        void main(){Out_Color=Frag_Color*texture(Texture,Frag_UV);}";

    public ImGuiController(GL gl, IWindow window, IInputContext input)
    {
        _gl = gl;
        _window = window;

        var io = ImGui.GetIO();
        io.BackendFlags |= ImGuiBackendFlags.RendererHasVtxOffset;

        CreateDeviceObjects();
        BindInput(input);
    }

    private void CreateDeviceObjects()
    {
        _vao = _gl.GenVertexArray();
        _vbo = _gl.GenBuffer();
        _ebo = _gl.GenBuffer();

        uint CompileShader(ShaderType type, string src)
        {
            var s = _gl.CreateShader(type);
            _gl.ShaderSource(s, src);
            _gl.CompileShader(s);
            return s;
        }

        var vert = CompileShader(ShaderType.VertexShader, VertSrc);
        var frag = CompileShader(ShaderType.FragmentShader, FragSrc);
        _shader = _gl.CreateProgram();
        _gl.AttachShader(_shader, vert);
        _gl.AttachShader(_shader, frag);
        _gl.LinkProgram(_shader);
        _gl.DeleteShader(vert);
        _gl.DeleteShader(frag);

        _uniformTex      = _gl.GetUniformLocation(_shader, "Texture");
        _uniformProjMtx  = _gl.GetUniformLocation(_shader, "ProjMtx");
        _attribVtxPos    = (uint)_gl.GetAttribLocation(_shader, "Position");
        _attribVtxUV     = (uint)_gl.GetAttribLocation(_shader, "UV");
        _attribVtxColor  = (uint)_gl.GetAttribLocation(_shader, "Color");

        // Upload font atlas (built by StubUiServices constructor)
        var io = ImGui.GetIO();
        io.Fonts.GetTexDataAsRGBA32(out byte* pixels, out int w, out int h);
        _fontTexture = _gl.GenTexture();
        _gl.BindTexture(TextureTarget.Texture2D, _fontTexture);
        _gl.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureMinFilter, (int)GLEnum.Linear);
        _gl.TexParameter(TextureTarget.Texture2D, TextureParameterName.TextureMagFilter, (int)GLEnum.Linear);
        _gl.TexImage2D(TextureTarget.Texture2D, 0, InternalFormat.Rgba, (uint)w, (uint)h, 0, PixelFormat.Rgba, PixelType.UnsignedByte, pixels);
        io.Fonts.SetTexID((IntPtr)_fontTexture);
        io.Fonts.ClearTexData();
    }

    private static void BindInput(IInputContext input)
    {
        var io = ImGui.GetIO();
        foreach (var mouse in input.Mice)
        {
            mouse.MouseMove  += (_, pos)    => io.MousePos = pos;
            mouse.MouseDown  += (_, btn)    => io.MouseDown[(int)btn] = true;
            mouse.MouseUp    += (_, btn)    => io.MouseDown[(int)btn] = false;
            mouse.Scroll     += (_, scroll) => { io.MouseWheelH += scroll.X; io.MouseWheel += scroll.Y; };
        }
        foreach (var kb in input.Keyboards)
            kb.KeyChar += (_, c) => io.AddInputCharacter(c);
    }

    public void NewFrame(float deltaTime)
    {
        var io = ImGui.GetIO();
        io.DisplaySize = new Vector2(_window.FramebufferSize.X, _window.FramebufferSize.Y);
        io.DeltaTime   = MathF.Max(deltaTime, 0.0001f);
        ImGui.NewFrame();
    }

    public void Render()
    {
        ImGui.Render();
        RenderDrawData(ImGui.GetDrawData());
    }

    private void RenderDrawData(ImDrawDataPtr data)
    {
        if (data.CmdListsCount == 0) return;

        _gl.Enable(EnableCap.Blend);
        _gl.BlendEquation(BlendEquationModeEXT.FuncAdd);
        _gl.BlendFunc(BlendingFactor.SrcAlpha, BlendingFactor.OneMinusSrcAlpha);
        _gl.Disable(EnableCap.CullFace);
        _gl.Disable(EnableCap.DepthTest);
        _gl.Enable(EnableCap.ScissorTest);

        var fb = _window.FramebufferSize;
        _gl.Viewport(0, 0, (uint)fb.X, (uint)fb.Y);

        float L = data.DisplayPos.X, R = data.DisplayPos.X + data.DisplaySize.X;
        float T = data.DisplayPos.Y, B = data.DisplayPos.Y + data.DisplaySize.Y;
        float[] proj =
        [
             2f/(R-L),  0,          0, 0,
             0,         2f/(T-B),   0, 0,
             0,         0,         -1, 0,
            (R+L)/(L-R),(T+B)/(B-T),0, 1,
        ];

        _gl.UseProgram(_shader);
        _gl.Uniform1(_uniformTex, 0);
        _gl.UniformMatrix4(_uniformProjMtx, 1, false, proj);
        _gl.BindVertexArray(_vao);

        for (int n = 0; n < data.CmdListsCount; n++)
        {
            var cmdList = data.CmdLists[n];

            _gl.BindBuffer(BufferTargetARB.ArrayBuffer, _vbo);
            _gl.BufferData(BufferTargetARB.ArrayBuffer,
                (nuint)(cmdList.VtxBuffer.Size * sizeof(ImDrawVert)),
                (void*)cmdList.VtxBuffer.Data, BufferUsageARB.StreamDraw);

            _gl.BindBuffer(BufferTargetARB.ElementArrayBuffer, _ebo);
            _gl.BufferData(BufferTargetARB.ElementArrayBuffer,
                (nuint)(cmdList.IdxBuffer.Size * sizeof(ushort)),
                (void*)cmdList.IdxBuffer.Data, BufferUsageARB.StreamDraw);

            _gl.EnableVertexAttribArray(_attribVtxPos);
            _gl.EnableVertexAttribArray(_attribVtxUV);
            _gl.EnableVertexAttribArray(_attribVtxColor);
            _gl.VertexAttribPointer(_attribVtxPos,   2, VertexAttribPointerType.Float,        false, (uint)sizeof(ImDrawVert), (void*)Marshal.OffsetOf<ImDrawVert>("pos"));
            _gl.VertexAttribPointer(_attribVtxUV,    2, VertexAttribPointerType.Float,        false, (uint)sizeof(ImDrawVert), (void*)Marshal.OffsetOf<ImDrawVert>("uv"));
            _gl.VertexAttribPointer(_attribVtxColor, 4, VertexAttribPointerType.UnsignedByte, true,  (uint)sizeof(ImDrawVert), (void*)Marshal.OffsetOf<ImDrawVert>("col"));

            for (int ci = 0; ci < cmdList.CmdBuffer.Size; ci++)
            {
                var cmd = cmdList.CmdBuffer[ci];
                _gl.BindTexture(TextureTarget.Texture2D, (uint)cmd.TextureId);

                var clip = new Vector4(
                    cmd.ClipRect.X - data.DisplayPos.X, cmd.ClipRect.Y - data.DisplayPos.Y,
                    cmd.ClipRect.Z - data.DisplayPos.X, cmd.ClipRect.W - data.DisplayPos.Y);
                _gl.Scissor((int)clip.X, (int)(fb.Y - clip.W), (uint)(clip.Z - clip.X), (uint)(clip.W - clip.Y));
                _gl.DrawElementsBaseVertex(PrimitiveType.Triangles, cmd.ElemCount,
                    DrawElementsType.UnsignedShort, (void*)(cmd.IdxOffset * sizeof(ushort)), (int)cmd.VtxOffset);
            }
        }

        _gl.Disable(EnableCap.ScissorTest);
        _gl.Disable(EnableCap.Blend);
    }

    public void Dispose()
    {
        _gl.DeleteVertexArray(_vao);
        _gl.DeleteBuffer(_vbo);
        _gl.DeleteBuffer(_ebo);
        _gl.DeleteProgram(_shader);
        _gl.DeleteTexture(_fontTexture);
    }
}

// UIStudio/StubUiServices.cs
using Craftimizer.Utils;
using ImGuiNET;
using System.Reflection;

namespace Craftimizer.UIStudio;

// Inicialize ANTES de criar o ImGuiController (precisa adicionar fontes ao atlas).
internal sealed class StubUiServices : IUiServices
{
    private ImFontPtr _iconFont;

    public float GlobalScale => 1.0f;
    public ImFontPtr IconFont => _iconFont;

    public StubUiServices()
    {
        var io = ImGui.GetIO();
        io.Fonts.AddFontDefault();

        var ttfBytes = LoadEmbeddedTtf();
        unsafe
        {
            fixed (byte* ptr = ttfBytes)
            {
                var cfg = ImGuiNative.ImFontConfig_ImFontConfig();
                cfg->MergeMode = 1;
                cfg->PixelSnapH = 1;

                // Range FA Solid: e000–f8ff
                var ranges = stackalloc ushort[] { 0xe000, 0xf8ff, 0 };
                _iconFont = io.Fonts.AddFontFromMemoryTTF((IntPtr)ptr, ttfBytes.Length, 14f, cfg, (IntPtr)ranges);
            }
        }
        // NÃO chame io.Fonts.Build() aqui — ImGuiController faz isso.
    }

    private static byte[] LoadEmbeddedTtf()
    {
        var asm = Assembly.GetExecutingAssembly();
        const string resourceName = "Craftimizer.UIStudio.Resources.fa-solid-900.ttf";
        using var stream = asm.GetManifestResourceStream(resourceName)
            ?? throw new FileNotFoundException($"Embedded resource not found: {resourceName}");
        using var ms = new MemoryStream();
        stream.CopyTo(ms);
        return ms.ToArray();
    }
}

// UIStudio/StudioApp.cs
using Craftimizer.Utils;
using ImGuiNET;
using Silk.NET.Maths;
using Silk.NET.OpenGL;
using Silk.NET.Windowing;
using System.Numerics;

namespace Craftimizer.UIStudio;

internal sealed class StudioApp : IDisposable
{
    private readonly IWindow _window;
    private GL _gl = null!;
    private ImGuiController _controller = null!;
    private StudioWindow _studioWindow = null!;

    public StudioApp()
    {
        var opts = WindowOptions.Default with
        {
            Size  = new Vector2D<int>(1280, 720),
            Title = "Craftimizer UI Studio",
        };
        _window = Window.Create(opts);
        _window.Load   += OnLoad;
        _window.Render += OnRender;
        _window.Closing += OnClosing;
    }

    private void OnLoad()
    {
        _gl = _window.CreateOpenGL();
        var input = _window.CreateInput();

        ImGui.CreateContext();
        var services = new StubUiServices(); // adiciona fontes antes do build
        UiServices.Current = services;

        _controller  = new ImGuiController(_gl, _window, input); // faz build do atlas
        _studioWindow = new StudioWindow();

        ApplyDarkTheme();
    }

    private static void ApplyDarkTheme()
    {
        var style = ImGui.GetStyle();
        style.WindowRounding  = 6f;
        style.FrameRounding   = 4f;
        style.ChildRounding   = 6f;
        ImGui.StyleColorsDark();
        // sobrescrever com as cores exatas do Craftimizer
        var colors = style.Colors;
        colors[(int)ImGuiCol.WindowBg]   = new Vector4(0.051f, 0.067f, 0.125f, 1f);
        colors[(int)ImGuiCol.ChildBg]    = new Vector4(0.078f, 0.098f, 0.157f, 1f);
        colors[(int)ImGuiCol.FrameBg]    = new Vector4(0.106f, 0.133f, 0.208f, 1f);
        colors[(int)ImGuiCol.FrameBgHovered] = new Vector4(0.129f, 0.165f, 0.251f, 1f);
        colors[(int)ImGuiCol.Border]     = new Vector4(0.392f, 0.549f, 0.784f, 0.30f);
    }

    private void OnRender(double dt)
    {
        _gl.ClearColor(0.024f, 0.031f, 0.063f, 1f); // BgBase
        _gl.Clear(ClearBufferMask.ColorBufferBit);

        _controller.NewFrame((float)dt);
        _studioWindow.Draw();
        _controller.Render();
    }

    private void OnClosing()
    {
        _controller.Dispose();
        ImGui.DestroyContext();
    }

    public void Run() => _window.Run();

    public void Dispose() => _window.Dispose();
}
```

// UIStudio/Storybook/StudioWindow.cs
using ImGuiNET;
using System.Numerics;
using System.Reflection;

namespace Craftimizer.UIStudio;

internal sealed class StudioWindow
{
    private readonly IReadOnlyList<IStory> _stories;
    private int _selectedStory;
    private int _selectedVariation;

    public StudioWindow()
    {
        _stories = DiscoverStories(Assembly.GetExecutingAssembly());
    }

    public static IReadOnlyList<IStory> DiscoverStories(Assembly asm) =>
        asm.GetTypes()
           .Where(t => t.IsClass && !t.IsAbstract && typeof(IStory).IsAssignableFrom(t))
           .Select(t => (IStory)Activator.CreateInstance(t)!)
           .OrderBy(s => s.Category)
           .ThenBy(s => s.Name)
           .ToList();

    public void Draw()
    {
        var viewport = ImGui.GetMainViewport();
        ImGui.SetNextWindowPos(viewport.Pos);
        ImGui.SetNextWindowSize(viewport.Size);
        ImGui.Begin("##studio", ImGuiWindowFlags.NoTitleBar | ImGuiWindowFlags.NoResize |
                                ImGuiWindowFlags.NoMove | ImGuiWindowFlags.NoBringToFrontOnFocus);

        DrawSidebar();
        ImGui.SameLine();
        DrawMain();

        ImGui.End();
    }

    private void DrawSidebar()
    {
        ImGui.BeginChild("##sidebar", new Vector2(200, 0), ImGuiChildFlags.Borders);

        ImGui.TextColored(new Vector4(0.32f, 0.90f, 0.63f, 1f), "UI Studio");
        ImGui.Separator();
        ImGui.Spacing();

        string? lastCategory = null;
        for (int i = 0; i < _stories.Count; i++)
        {
            var story = _stories[i];
            if (story.Category != lastCategory)
            {
                if (lastCategory != null) ImGui.Spacing();
                ImGui.TextColored(new Vector4(0.31f, 0.38f, 0.48f, 1f), story.Category.ToUpperInvariant());
                lastCategory = story.Category;
            }

            var selected = _selectedStory == i;
            if (ImGui.Selectable($"  {story.Name}", selected))
            {
                _selectedStory    = i;
                _selectedVariation = 0;
            }
        }

        ImGui.EndChild();
    }

    private void DrawMain()
    {
        if (_stories.Count == 0)
        {
            ImGui.TextDisabled("Nenhuma story encontrada.");
            return;
        }

        var story = _stories[_selectedStory];

        ImGui.BeginChild("##main", Vector2.Zero, ImGuiChildFlags.None);

        // Story tabs (variações)
        var names = story.StoryNames;
        for (int i = 0; i < names.Length; i++)
        {
            if (i > 0) ImGui.SameLine();
            var active = _selectedVariation == i;
            if (active) ImGui.PushStyleColor(ImGuiCol.Button, new Vector4(0.32f, 0.90f, 0.63f, 0.3f));
            if (ImGui.SmallButton(names[i])) _selectedVariation = i;
            if (active) ImGui.PopStyleColor();
        }
        ImGui.Separator();

        // Preview + Controls split
        var availW = ImGui.GetContentRegionAvail().X;
        var controlsW = 220f;
        var previewW = availW - controlsW - ImGui.GetStyle().ItemSpacing.X;

        // Preview
        ImGui.BeginChild("##preview", new Vector2(previewW, 0), ImGuiChildFlags.Borders);
        story.DrawPreview(_selectedVariation);
        ImGui.EndChild();

        ImGui.SameLine();

        // Controls
        ImGui.BeginChild("##controls", new Vector2(controlsW, 0), ImGuiChildFlags.Borders);
        ImGui.TextColored(new Vector4(0.31f, 0.38f, 0.48f, 1f), "CONTROLS");
        ImGui.Separator();
        story.DrawControls(_selectedVariation);
        ImGui.EndChild();

        ImGui.EndChild();
    }
}

using Craftimizer.Utils;

namespace Craftimizer.UIStudio.Stories.Atoms;

// UIStudio/Stories/Molecules/ChartsStory.cs
using Craftimizer.Utils;
using ImGuiNET;
using System.Numerics;

namespace Craftimizer.UIStudio.Stories.Molecules;

internal sealed class ChartsStory : IStory
{
    public string Name => "Charts";
    public string Category => "Molecules";
    public string[] StoryNames => ["stat arcs — inline", "stat arcs — grid", "arc color variants"];

    private float _progress    = 0.72f;
    private float _quality     = 0.45f;
    private float _durability  = 0.90f;
    private float _cp          = 0.33f;

    public void DrawControls(int idx)
    {
        ImGui.SliderFloat("progress",   ref _progress,   0f, 1f);
        ImGui.SliderFloat("quality",    ref _quality,    0f, 1f);
        ImGui.SliderFloat("durability", ref _durability, 0f, 1f);
        ImGui.SliderFloat("cp",         ref _cp,         0f, 1f);
    }

    public void DrawPreview(int idx)
    {
        var dl = ImGui.GetWindowDrawList();
        const float size = 80f;
        var spacing = ImGui.GetStyle().ItemSpacing.X;

        switch (idx)
        {
            case 0: // inline (4 arcos em linha)
            {
                var origin = ImGui.GetCursorScreenPos();
                ImGui.Dummy(new Vector2(size * 4 + spacing * 3, size));
                DrawArc(dl, origin,                                   size, _progress,   Colors.Progress,   "Progress");
                DrawArc(dl, origin + new Vector2(size + spacing, 0),  size, _quality,    Colors.Quality,    "Quality");
                DrawArc(dl, origin + new Vector2((size + spacing)*2, 0), size, _durability, Colors.Durability, "Durability");
                DrawArc(dl, origin + new Vector2((size + spacing)*3, 0), size, _cp,         Colors.CP,         "CP");
                break;
            }
            case 1: // grid 2×2
            {
                var origin = ImGui.GetCursorScreenPos();
                ImGui.Dummy(new Vector2(size * 2 + spacing, size * 2 + spacing));
                DrawArc(dl, origin,                                     size, _progress,   Colors.Progress,   "Progress");
                DrawArc(dl, origin + new Vector2(size + spacing, 0),    size, _quality,    Colors.Quality,    "Quality");
                DrawArc(dl, origin + new Vector2(0, size + spacing),    size, _durability, Colors.Durability, "Durability");
                DrawArc(dl, origin + new Vector2(size + spacing, size + spacing), size, _cp, Colors.CP, "CP");
                break;
            }
            case 2: // variações de cor com a mesma fração
            {
                var origin = ImGui.GetCursorScreenPos();
                ImGui.Dummy(new Vector2(size * 5 + spacing * 4, size));
                var all = new[] { (Colors.Progress, "Progress"), (Colors.Quality, "Quality"), (Colors.Durability, "Durability"), (Colors.CP, "CP"), (Colors.Collectability, "Collect.") };
                for (int i = 0; i < all.Length; i++)
                    DrawArc(dl, origin + new Vector2(i * (size + spacing), 0), size, _progress, all[i].Item1, all[i].Item2);
                break;
            }
        }
    }

    private static void DrawArc(ImDrawListPtr dl, Vector2 pos, float size, float frac, Vector4 color, string label)
    {
        ImGuiUtils.DrawStatArc(dl, pos, size, frac, color);
        if (ImGui.IsMouseHoveringRect(pos, pos + new Vector2(size)))
            ImGui.SetTooltip($"{label}: {frac:P0}");
    }
}
